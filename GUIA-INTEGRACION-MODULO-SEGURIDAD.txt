 * import { AutenticacionServicio } from '../../servicios/autenticacion.servicio';
 * 
 * export class OtroComponente {
 *   usuarioActual: Usuario | null = null;
 *   
 *   constructor(private auth: AutenticacionServicio) {}
 *   
 *   ngOnInit() {
 *     // Obtener usuario actual
 *     this.usuarioActual = this.auth.obtenerUsuarioActual();
 *     
 *     // Suscribirse a cambios
 *     this.auth.usuarioAutenticado$.subscribe(usuario => {
 *       this.usuarioActual = usuario;
 *     });
 *   }
 *   
 *   estaAutenticado(): boolean {
 *     return this.usuarioActual !== null;
 *   }
 * }

 * =============================================================================
 * PASO 4: CREAR UN GUARD DE AUTENTICACIÓN (RECOMENDADO)
 * =============================================================================
 * 
 * import { Injectable } from '@angular/core';
 * import { CanActivate, Router } from '@angular/router';
 * import { AutenticacionServicio } from './servicios/autenticacion.servicio';
 * 
 * @Injectable({
 *   providedIn: 'root'
 * })
 * export class GuardAutenticacion implements CanActivate {
 *   constructor(
 *     private auth: AutenticacionServicio,
 *     private router: Router
 *   ) {}
 *   
 *   canActivate(): boolean {
 *     const usuario = this.auth.obtenerUsuarioActual();
 *     if (usuario) {
 *       return true;
 *     } else {
 *       this.router.navigate(['/login']);
 *       return false;
 *     }
 *   }
 * }
 * 
 * // Usar en rutas:
 * {
 *   path: 'dashboard',
 *   component: DashboardComponent,
 *   canActivate: [GuardAutenticacion]
 * }
 * 
 * =============================================================================
 * PASO 5: CREAR UN GUARD DE ROLES (RECOMENDADO PARA FUTURO)
 * =============================================================================
 * 
 * import { Injectable } from '@angular/core';
 * import { CanActivate, Router, ActivatedRouteSnapshot } from '@angular/router';
 * import { AutenticacionServicio } from './servicios/autenticacion.servicio';
 * 
 * @Injectable({
 *   providedIn: 'root'
 * })
 * export class GuardRol implements CanActivate {
 *   constructor(
 *     private auth: AutenticacionServicio,
 *     private router: Router
 *   ) {}
 *   
 *   canActivate(route: ActivatedRouteSnapshot): boolean {
 *     const usuario = this.auth.obtenerUsuarioActual();
 *     const rolesRequeridos = route.data['roles'] as string[];
 *     
 *     if (usuario && rolesRequeridos.includes(usuario.rol.tipo)) {
 *       return true;
 *     } else {
 *       this.router.navigate(['/acceso-denegado']);
 *       return false;
 *     }
 *   }
 * }
 * 
 * // Usar en rutas:
 * {
 *   path: 'admin',
 *   component: AdminComponent,
 *   canActivate: [GuardRol],
 *   data: { roles: ['administrador'] }
 * }
