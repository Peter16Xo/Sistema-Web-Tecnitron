 * CÓMO USAR EL MÓDULO
 * ===============================================================================
 * 
 * 1. IMPORTAR EN COMPONENTE:
 *    import { LoginComponent } from './modulos/seguridad/componentes/login/login.component';
 *    
 *    @Component({
 *      imports: [LoginComponent],
 *      template: '<app-login></app-login>'
 *    })
 * 
 * 2. INYECTAR SERVICIO:
 *    constructor(private auth: AutenticacionServicio) {}
 *    
 * 3. VERIFICAR AUTENTICACIÓN:
 *    if (this.auth.obtenerUsuarioActual()) {
 *      // Usuario autenticado
 *    }
 * 
 * 4. SUSCRIBIRSE A CAMBIOS:
 *    this.auth.usuarioAutenticado$.subscribe(usuario => {
 *      // Reaccionar a cambios
 *    });
